rustup toolchain install nightly
rustup target add aarch64-unknown-none --toolchain nightly


# 构建用户态服务与工具
cargo run -p xtask -- build-servers
cargo run -p xtask -- build-installer
cargo run -p xtask -- build-recovery

# 说明：build-servers 现在会同时产出 zero-launchd、zero-securityd、
# zero-blkdrv 与 zero-fsd 的 aarch64-unknown-none 裸机 release ELF，
# 并经 sync_userland_blobs 同步到 kernel/rootfs/System/Core/
# （zero-launchd / zero-securityd / zero-blkdrv / zero-fsd），随后的
# build-rootfs / build-iso 会把它们嵌入内核镜像。
#
# 第七刀文件服务上线（fsd + blkdrv）：
#   - blkdrv：块设备服务，数据面走内核直通（号位 7/8 BlockRead/
#     BlockWrite），协议见 zero_abi::protocol::blk::passthrough；
#   - fsd：块后端卷文件服务（名字→LBA 内存表，超级块 @LBA8192，
#     数据区 @LBA16384 起；避开内核自检@2048 与 userdb@4096 起），
#     协议见 zero_abi::protocol::fs::vtable；
#   - 两者由 launchd 开机自动对账拉起（KNOWN_SERVICES 已含）。

# 构建微内核/裸机内核
cargo run -p xtask -- build-kernel

cargo run -p xtask -- build-rootfs

cargo run -p zero-installer


# 收集内核、rootfs、工具并打包成 ISO
cargo run -p xtask -- build-iso
#   ↑ build-iso 入口已前置 build_baremetal_blobs：自动先编最新
#   zero-launchd / zero-securityd 裸机版，再 sync → build-rootfs →
#   installer → 内核嵌入，全自动产出含新服务的 ISO，无手动步骤。

# ---- QEMU 实机运行（virtio-blk 挂盘）----

# 1) 创建 32MiB raw 测试盘（已存在则跳过；xtask run 与 run-qemu.sh 会自动创建）
dd if=/dev/zero of=target/disk.img bs=1m count=32

# 2) 一键启动（推荐二选一）：
#    a) xtask run：串口写 target/qemu-run.log，适合脚本化验证
cargo run -p xtask -- run            # 正常模式（自动挂盘 target/disk.img）
cargo run -p xtask -- run recovery   # 恢复模式（-boot d）
#
#    b) run-qemu.sh：交互串口 mon:stdio，可直接操作用户态 shell（Ctrl-A X 退出）
./run-qemu.sh

# 3) 等价手动命令（与上述脚本参数一致）：
#    槽位约定：virtio-mmio-bus.0 @ 0x0a000000 = 驱动表 virtio-blk0，GIC INTID 48。
qemu-system-aarch64 \
  -machine virt \
  -cpu cortex-a72 \
  -m 1024 \
  -bios /opt/homebrew/share/qemu/edk2-aarch64-code.fd \
  -serial mon:stdio \
  -display none \
  -cdrom target/zero-os.iso \
  -drive if=none,file=target/disk.img,format=raw,id=zdisk \
  -device virtio-blk-device,drive=zdisk,bus=virtio-mmio-bus.0

# 注意：测试盘必须用专用空盘 target/disk.img；
# 不要挂 zero-rootfs.img——用户态（如 securityd 的 userdb）会真实写入该盘。

# ---- 用户态服务生命周期（launchd）----
#
# 当前内核登记并交由 launchd 拉起的服务：securityd、blkdrv、fsd
# （均 /System/Core/ 下裸机 ELF；blkdrv privileged=true 预留 MMIO 租约，
# fsd 纯 IPC。blkdrv 缺盘时降级告警；fsd 探测不到 blkdrv 时同样降级）。
#
# 开机自动对账：launchd 启动时逐个向内核 spawn KNOWN_SERVICES——
# 已登记的服务记为 [running pid=N]，未登记的保持 [not-installed]。
# 因此启动日志可见：fsd/blkdrv/securityd banner + launchd 对账行
# （"launchd: service online: <name> pid=N"）。
#
# shell 验证命令：
#   launchd list            # 应看到 fsd/blkdrv/securityd 均 [running] pid=N
#   launchd spawn securityd # 已运行 → 友好提示 "already running pid=N"（幂等，不重复拉起）
#   fstest                  # 文件语义端到端验证：
#                           #   open(create)→write "ZERO-FSD"→close→
#                           #   reopen→read 比对→close，打印 PASS/MATCH
#   launchd status          # launchd up; services registered=N
#
# 脚本化实机验证（无盘、串口注入两条命令，约 26 秒后看 /tmp/svc.log）：
#   ( sleep 14; printf 'launchd list\r'; sleep 3; printf 'launchd spawn securityd\r'; sleep 3 ) \
#     | qemu-system-aarch64 -machine virt -cpu cortex-a72 -m 1024 \
#       -bios /opt/homebrew/share/qemu/edk2-aarch64-code.fd \
#       -serial mon:stdio -display none -cdrom target/zero-os.iso > /tmp/svc.log 2>&1 &
