rustup toolchain install nightly
rustup target add aarch64-unknown-none --toolchain nightly


# 构建用户态服务与工具
cargo run -p xtask -- build-servers
cargo run -p xtask -- build-installer
cargo run -p xtask -- build-recovery

# 构建微内核/裸机内核
cargo run -p xtask -- build-kernel
cargo +nightly build \
  -Z build-std=core,compiler_builtins,alloc \
  --target aarch64-unknown-none \
  -p zero-kernel


# 构建 Zero OS UEFI 引导程序
cargo +nightly build \
  -Z build-std=core,compiler_builtins,alloc \
  --target aarch64-unknown-uefi \
  -p bootloader-uefi


# 把 kernel/rootfs 目录打包为 bundle
cargo run -p xtask -- build-rootfs

# 调用安装器，用 mkzfs 写出 rootfs 镜像
cargo run -p zero-installer


# 收集内核、rootfs、工具并打包成 ISO
cargo run -p xtask -- build-iso

# Zero OS 自研 UEFI Loader 默认 0 秒进入系统，连续失败 3 次自动进入恢复。
# Zero OS 成功启动后请在系统里执行 `bootctl`
# （默认操作 /boot/EFI/ZEROOS/boot.env）清除失败计数。


# ---- QEMU 实机运行（virtio-blk 挂盘）----

# 1) 创建 32MiB raw 测试盘（已存在则跳过；xtask run 与 run-qemu.sh 会自动创建）
dd if=/dev/zero of=target/disk.img bs=1m count=32

# 2) 一键启动（推荐二选一）：
#    a) xtask run：串口写 target/qemu-run.log，适合脚本化验证
cargo run -p xtask -- run            # 正常模式（自动挂盘 target/disk.img）
cargo run -p xtask -- run recovery   # 恢复模式（-boot d）
#
#    b) run-qemu.sh：交互串口 mon:stdio，可直接操作用户态 shell（Ctrl-A X 退出）
./run-qemu.sh

# 3) 等价手动命令（与上述脚本参数一致）：
#    槽位约定：virtio-mmio-bus.0 @ 0x0a000000 = 驱动表 virtio-blk0，GIC INTID 48。
qemu-system-aarch64 \
  -machine virt \
  -cpu cortex-a72 \
  -m 1024 \
  -bios /opt/homebrew/share/qemu/edk2-aarch64-code.fd \
  -serial mon:stdio \
  -display none \
  -cdrom target/zero-os.iso \
  -drive if=none,file=target/disk.img,format=raw,id=zdisk \
  -device virtio-blk-device,drive=zdisk,bus=virtio-mmio-bus.0

# 注意：测试盘必须用专用空盘 target/disk.img；
# 不要挂 zero-rootfs.img——用户态（如 securityd 的 userdb）会真实写入该盘。
